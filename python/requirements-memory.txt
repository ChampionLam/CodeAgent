# sidecar.py 的 _memory_hooks() 用 importlib.util.find_spec("mnemosyne")

# extras 口径:
#   [mcp]        —— MCP stdio server 子进程(sidecar 用

#   [embeddings] —— fastembed>=0.3.0 + sqlite-vec>=0.1.0(本地 ONNX 嵌入)
#   明确不要 [llm] extra(llama-cpp/ctransformers 等):宿主已把

# fastembed 经 HF_ENDPOINT 镜像懒加载(宿主 python/memory/memory_config.py
# 的 build_env() 负责注入缓存目录等环境变量)。
#

#   mnemosyne-memory==3.15.1  fastembed==0.9.0
